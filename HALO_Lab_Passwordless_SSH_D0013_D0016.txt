HALO LAB — PASSWORDLESS SSH SETUP FOR D0013–D0016

# 1. Confirm the existing HALO SSH key is present.
ls -l ~/.ssh/halo_archive_ed25519 ~/.ssh/halo_archive_ed25519.pub

# 2. Create a dedicated SSH alias file for the five lab drones.
mkdir -p ~/.ssh
chmod 700 ~/.ssh

cat > ~/.ssh/halo_archive_config <<'EOF'
Host halo-d0012
    HostName 192.168.0.20
    User root
    IdentityFile ~/.ssh/halo_archive_ed25519
    IdentitiesOnly yes
    ServerAliveInterval 10

Host halo-d0013
    HostName 192.168.0.21
    User root
    IdentityFile ~/.ssh/halo_archive_ed25519
    IdentitiesOnly yes
    ServerAliveInterval 10

Host halo-d0014
    HostName 192.168.0.22
    User root
    IdentityFile ~/.ssh/halo_archive_ed25519
    IdentitiesOnly yes
    ServerAliveInterval 10

Host halo-d0015
    HostName 192.168.0.23
    User root
    IdentityFile ~/.ssh/halo_archive_ed25519
    IdentitiesOnly yes
    ServerAliveInterval 10

Host halo-d0016
    HostName 192.168.0.24
    User root
    IdentityFile ~/.ssh/halo_archive_ed25519
    IdentitiesOnly yes
    ServerAliveInterval 10
EOF

chmod 600 ~/.ssh/halo_archive_config

# 3. Make OpenSSH load the dedicated HALO alias file.
touch ~/.ssh/config
chmod 600 ~/.ssh/config

grep -Fxq 'Include /home/zafar/.ssh/halo_archive_config' ~/.ssh/config || {
    cp ~/.ssh/config ~/.ssh/config.before_halo_swarm
    {
        echo 'Include /home/zafar/.ssh/halo_archive_config'
        cat ~/.ssh/config.before_halo_swarm
    } > ~/.ssh/config
}

# 4. Confirm each alias resolves to the correct lab IP and key.
ssh -G halo-d0013 | grep -E '^(hostname|user|identityfile) '
ssh -G halo-d0014 | grep -E '^(hostname|user|identityfile) '
ssh -G halo-d0015 | grep -E '^(hostname|user|identityfile) '
ssh -G halo-d0016 | grep -E '^(hostname|user|identityfile) '

# 5. Install the HALO public key on D0013.
# When prompted, enter the common drone password you already have.
ssh-copy-id -i ~/.ssh/halo_archive_ed25519.pub root@192.168.0.21

# 6. Install the HALO public key on D0014.
# When prompted, enter the common drone password you already have.
ssh-copy-id -i ~/.ssh/halo_archive_ed25519.pub root@192.168.0.22

# 7. Install the HALO public key on D0015.
# When prompted, enter the common drone password you already have.
ssh-copy-id -i ~/.ssh/halo_archive_ed25519.pub root@192.168.0.23

# 8. Install the HALO public key on D0016.
# When prompted, enter the common drone password you already have.
ssh-copy-id -i ~/.ssh/halo_archive_ed25519.pub root@192.168.0.24

# 9. Verify passwordless SSH to all five drones.
for D in d0012 d0013 d0014 d0015 d0016; do
    echo "===== halo-$D ====="
    ssh -o BatchMode=yes -o ConnectTimeout=8 "halo-$D" "hostname && date -u"
done

# 10. Final pass/fail check. No password prompt is allowed.
for D in d0012 d0013 d0014 d0015 d0016; do
    ssh -o BatchMode=yes -o ConnectTimeout=8 "halo-$D" "true" || {
        echo "FAILED: passwordless SSH for halo-$D"
        exit 1
    }
done

echo "PASSWORDLESS SSH READY FOR D0012-D0016"
