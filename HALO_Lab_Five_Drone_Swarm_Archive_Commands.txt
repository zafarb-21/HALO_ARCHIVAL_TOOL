python3 - <<'PY'
import os
from pathlib import Path
import yaml

rows = {}
for line in Path(os.environ["HALO_DOMAIN_FILE"]).read_text().splitlines():
    drone_id, host, ip, domain = line.split()
    rows[drone_id] = {"host": host, "ip": ip, "domain": int(domain)}

cfg = yaml.safe_load(Path("drones/swarm_lab.yaml").read_text()) or {}
entries = cfg.get("drones")

if isinstance(entries, dict):
    converted = []
    for key, value in entries.items():
        item = dict(value or {})
        item.setdefault("drone_id", key)
        converted.append(item)
    entries = converted

if not isinstance(entries, list):
    raise SystemExit("ERROR: swarm_lab.yaml has no drones list/dict")

by_id = {str(x.get("drone_id")): x for x in entries if isinstance(x, dict)}
errors = []

for drone_id, expected in rows.items():
    item = by_id.get(drone_id)
    if not item:
        errors.append(f"{drone_id}: missing from swarm_lab.yaml")
        continue

    cfg_ip = str(item.get("ip_address", ""))
    cfg_host = str(item.get("ssh_host", item.get("host", "")))
    try:
        cfg_domain = int(item.get("ros_domain_id"))
    except Exception:
        cfg_domain = None

    if cfg_ip != expected["ip"]:
        errors.append(f"{drone_id}: config IP={cfg_ip!r}, live lab IP={expected['ip']!r}")
    if cfg_host and cfg_host != expected["host"]:
        errors.append(f"{drone_id}: config host={cfg_host!r}, expected={expected['host']!r}")
    if cfg_domain != expected["domain"]:
        errors.append(f"{drone_id}: config domain={cfg_domain!r}, live domain={expected['domain']!r}")
    if item.get("enabled") is False:
        errors.append(f"{drone_id}: enabled=false")
    if item.get("required") is False:
        errors.append(f"{drone_id}: required=false")

    try:
        post = float(item.get("post_landing_record_s", 3))
        if not 2.0 <= post <= 5.0:
            errors.append(f"{drone_id}: post_landing_record_s={post}, expected 2..5")
    except Exception:
        errors.append(f"{drone_id}: invalid post_landing_record_s")

if errors:
    for e in errors:
        print("ERROR:", e)
    raise SystemExit(1)

print("swarm_lab.yaml matches all five live lab drones.")
PY
